cmake_minimum_required(VERSION 3.20)
project(MagneticTorusDose LANGUAGES CXX)
find_package(Geant4 11.3 REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
include(${Geant4_USE_FILE})
add_executable(torus-dose dose_worker.cc)
target_compile_features(torus-dose PRIVATE cxx_std_17)
target_link_libraries(torus-dose PRIVATE ${Geant4_LIBRARIES} nlohmann_json::nlohmann_json)
install(TARGETS torus-dose RUNTIME DESTINATION bin)
